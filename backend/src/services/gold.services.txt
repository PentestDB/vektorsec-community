import SessionsModel, { SessionDoc } from "../models/Sessions/Sessions.model";
import UserModel, { UserDoc } from "../models/User/User.model";

export const addGoldForUser = async ({
  user,
  gold,
  description,
}: {
  user: UserDoc;
  gold: number;
  description: string;
}) => {
  try {
    if (!user) {
      throw new Error("User not found");
    }

    user.gold = user.gold + Number(gold);
    user.goldTransactions = [
      ...user.goldTransactions,
      {
        goldTransacted: gold,
        type: "credit",
        description,
        date: new Date(),
      },
    ];

    await user.save();

    return user;
  } catch (error: any) {
    throw new Error(error);
  }
};

export const deductGoldForUser = async ({
  user,
  gold,
  description,
}: {
  user: UserDoc;
  gold: number;
  description: string;
}) => {
  try {
    if (!user) {
      throw new Error("User not found");
    }

    user.gold = user.gold - Number(gold);
    user.goldTransactions = [
      ...user.goldTransactions,
      {
        goldTransacted: gold,
        type: "debit",
        description,
        date: new Date(),
      },
    ];

    await user.save();

    return user;
  } catch (error: any) {
    throw new Error(error);
  }
};

export const checkIfUserEnoughGold = async (sessionId: string) => {
  try {
    const session = await SessionsModel.findOne({ sessionId });

    if (!session) {
      throw new Error("Session not found");
    }

    const user = await UserModel.findById(session.uid);

    if (!user) {
      throw new Error("User not found");
    }

    if (user.gold < 1 && user.billing.plan !== "FLEXIBLE") {
      throw new Error("User does not have enough gold");
    }

    return true;

  } catch (error: any) {
    throw new Error(error);
  }
};
